  \providecommand{\bfr}[3]{\node[font=\scriptsize\bfseries, anchor=west, text=#2, inner sep=1pt] at ($(#1)+(0.38,0.22)$) {BF$=#3$};}
  \providecommand{\bfl}[3]{\node[font=\scriptsize\bfseries, anchor=east, text=#2, inner sep=1pt] at ($(#1)+(-0.38,0.22)$) {BF$=#3$};}
